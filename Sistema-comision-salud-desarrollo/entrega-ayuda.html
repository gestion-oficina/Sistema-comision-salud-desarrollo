function exportarAWord() { // Selecciona la tabla que contiene los datos de las entregas const tabla = document.getElementById("tabla-entregas"); // Asegúrate de que el id de tu tabla sea este o cámbialo if (!tabla) { alert("No se encontró la tabla de datos para exportar."); return; } // Estilos básicos para que el documento mantenga formato en Word const estiloMime = 'data:application/vnd.ms-word;charset=utf-8,'; const plantillaHtml = ` html, body, table, th, td { font-family: Arial, sans-serif; font-size: 11pt; } table { border-collapse: collapse; width: 100%; } th, td { border: 1px solid #d3d3d3; padding: 8px; text-align: left; } th { background-color: #1e3a8a; color: #ffffff; } `; const contenidoCompleto = ` Reporte de Entregas de Ayuda

Control de Entregas de Ayudas Sociales

Fecha de generación: ${new Date().toLocaleDateString()}

${tabla.outerHTML} `; // Crea un enlace virtual de descarga const blob = new Blob(['\ufeff' + contenidoCompleto], { type: 'application/msword' }); const url = URL.createObjectURL(blob); const etiquetaDescarga = document.createElement('a'); etiquetaDescarga.href = url; etiquetaDescarga.download = `Reporte_Entregas_Ayuda_${new Date().toISOString().slice(0, 10)}.doc`; document.body.appendChild(etiquetaDescarga); etiquetaDescarga.click(); document.body.removeChild(etiquetaDescarga); }
