import { SQLiteCloud } from '@sqlitecloud/drivers'; export default async function handler(req, res) { if (req.method !== 'POST') { return res.status(405).json({ error: 'Método no permitido' }); } const { email } = req.body; if (!email) { return res.status(400).json({ error: 'Falta el correo' }); } try { const sqlString = process.env.SQLITE_CLOUD_CONNECTION_STRING; const db = new SQLiteCloud(sqlString); const resultados = await db.sql` SELECT id, nombre, rol, coordinacion FROM usuarios_sistema WHERE LOWER(email) = LOWER(${email}); `; if (!resultados || resultados.length === 0) { return res.status(404).json({ error: 'Correo institucional no registrado' }); } return res.status(200).json({ usuario: resultados[0] }); } catch (error) { console.error('Error en base de datos:', error); return res.status(500).json({ error: 'Error de conexión con el sistema' }); } }
